cmake_minimum_required(VERSION 3.20)
project(IQMLuminanceInput LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(IQM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(iqm_luminance STATIC "${IQM_ROOT}/src/hdrvdp3/luminance.cpp")
target_include_directories(iqm_luminance PUBLIC "${IQM_ROOT}/include")
add_executable(luminance_input_test "${IQM_ROOT}/tests/verify_luminance_input.cpp")
target_link_libraries(luminance_input_test PRIVATE iqm_luminance)
add_executable(prepare_hdr_luminance "${IQM_ROOT}/tests/prepare_hdr_luminance.cpp")
target_include_directories(prepare_hdr_luminance PRIVATE "${IQM_ROOT}/lib/stb")
foreach(target IN ITEMS iqm_luminance luminance_input_test prepare_hdr_luminance)
    if(MSVC)
        target_compile_options(${target} PRIVATE /fp:strict /EHsc)
    else()
        target_compile_options(${target} PRIVATE -ffp-contract=off)
    endif()
endforeach()
include(CTest)
set(IQM_LUMINANCE_FIXTURE "" CACHE PATH "Fresh PFM input fixtures")
if(BUILD_TESTING AND IQM_LUMINANCE_FIXTURE)
    add_test(NAME luminance_pfm COMMAND luminance_input_test "${IQM_LUMINANCE_FIXTURE}")
endif()
