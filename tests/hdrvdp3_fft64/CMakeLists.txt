cmake_minimum_required(VERSION 3.20)
project(IQMFFT64Reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
get_filename_component(IQM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(fft64-reference-test ../verify_fft64_reference.cpp
    "${IQM_ROOT}/bin/shared/vulkan_res.cpp"
    "${IQM_ROOT}/bin/IQM/vulkan_instance.cpp"
    "${IQM_ROOT}/src/base/vulkan_runtime.cpp")
target_compile_definitions(fft64-reference-test PRIVATE VULKAN_HPP_NO_CONSTRUCTORS)
target_include_directories(fft64-reference-test PRIVATE
    "${IQM_ROOT}/include" "${IQM_ROOT}/bin/shared" "${IQM_ROOT}/bin/IQM")
target_include_directories(fft64-reference-test SYSTEM PRIVATE "${IQM_ROOT}/lib/stb")
target_link_libraries(fft64-reference-test PRIVATE Vulkan::Vulkan)
if(MSVC)
    target_compile_options(fft64-reference-test PRIVATE /EHsc /fp:strict)
else()
    target_compile_options(fft64-reference-test PRIVATE -ffp-contract=off)
endif()
if(WIN32)
    add_executable(fftw-plan-test ../inspect_fftw_plan.cpp)
    if(MSVC)
        target_compile_options(fftw-plan-test PRIVATE /EHsc)
    endif()
endif()
