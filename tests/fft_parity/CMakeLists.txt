cmake_minimum_required(VERSION 3.29)
project(IQMFFTParity LANGUAGES CXX)

# Independent validation tools: do not configure or link experimental metrics.
find_package(Vulkan REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(IQM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(IQM-base-noise-test ../verify_base_noise.cpp "${IQM_ROOT}/src/hdrvdp3/csf.cpp")
target_include_directories(IQM-base-noise-test PRIVATE "${IQM_ROOT}/include")
if(MSVC)
    target_compile_options(IQM-base-noise-test PRIVATE /fp:strict /EHsc)
else()
    target_compile_options(IQM-base-noise-test PRIVATE -ffp-contract=off)
endif()
add_executable(IQM-map-gpu-test ../verify_map_gpu.cpp
    "${IQM_ROOT}/src/hdrvdp3/map_gpu.cpp" "${IQM_ROOT}/src/hdrvdp3/csf.cpp" "${IQM_ROOT}/src/hdrvdp3/resize.cpp" "${IQM_ROOT}/src/hdrvdp3/quality.cpp"
    "${IQM_ROOT}/bin/shared/vulkan_res.cpp" "${IQM_ROOT}/bin/IQM/vulkan_instance.cpp" "${IQM_ROOT}/src/base/vulkan_runtime.cpp")
target_compile_definitions(IQM-map-gpu-test PRIVATE VULKAN_HPP_NO_CONSTRUCTORS)
target_include_directories(IQM-map-gpu-test PRIVATE "${IQM_ROOT}/include" "${IQM_ROOT}/bin/shared" "${IQM_ROOT}/bin/IQM")
target_include_directories(IQM-map-gpu-test SYSTEM PRIVATE "${IQM_ROOT}/lib/stb")
target_link_libraries(IQM-map-gpu-test PRIVATE Vulkan::Vulkan)
if(MSVC)
    target_compile_options(IQM-map-gpu-test PRIVATE /fp:strict /EHsc)
else()
    target_compile_options(IQM-map-gpu-test PRIVATE -ffp-contract=off)
endif()
add_library(IQM-jnd-lut-fixture OBJECT "${IQM_ROOT}/src/hdrvdp3/jnd.cpp")
target_sources(IQM-map-gpu-test PRIVATE $<TARGET_OBJECTS:IQM-jnd-lut-fixture>)
target_include_directories(IQM-jnd-lut-fixture PRIVATE "${IQM_ROOT}/include")
add_executable(IQM-jnd-knot-test ../verify_jnd_knots.cpp "${IQM_ROOT}/src/hdrvdp3/jnd_single.cpp" $<TARGET_OBJECTS:IQM-jnd-lut-fixture>)
target_include_directories(IQM-jnd-knot-test PRIVATE "${IQM_ROOT}/include")
if(MSVC)
    target_compile_options(IQM-jnd-knot-test PRIVATE /fp:strict /EHsc)
else()
    target_compile_options(IQM-jnd-knot-test PRIVATE -ffp-contract=off)
endif()
add_executable(IQM-masked-difference-diagnostic ../diagnose_masked_difference.cpp "${IQM_ROOT}/src/hdrvdp3/quality.cpp")
target_include_directories(IQM-masked-difference-diagnostic PRIVATE "${IQM_ROOT}/include")
if(MSVC)
    target_compile_options(IQM-masked-difference-diagnostic PRIVATE /fp:strict /EHsc)
else()
    target_compile_options(IQM-masked-difference-diagnostic PRIVATE -ffp-contract=off)
endif()
add_executable(IQM-resize-reference-test ../verify_resize_reference.cpp "${IQM_ROOT}/src/hdrvdp3/resize.cpp")
target_include_directories(IQM-resize-reference-test PRIVATE "${IQM_ROOT}/include")
if(MSVC)
    target_compile_options(IQM-resize-reference-test PRIVATE /fp:strict /EHsc)
else()
    target_compile_options(IQM-resize-reference-test PRIVATE -ffp-contract=off)
endif()
add_executable(IQM-csf-reference-test ../verify_csf_reference.cpp "${IQM_ROOT}/src/hdrvdp3/csf.cpp")
add_executable(IQM-csf-lut-reference-test ../verify_csf_lut_reference.cpp "${IQM_ROOT}/src/hdrvdp3/csf.cpp")
target_include_directories(IQM-csf-lut-reference-test PRIVATE "${IQM_ROOT}/include")
if(MSVC)
    target_compile_options(IQM-csf-lut-reference-test PRIVATE /fp:strict /EHsc)
else()
    target_compile_options(IQM-csf-lut-reference-test PRIVATE -ffp-contract=off)
endif()
target_include_directories(IQM-csf-reference-test PRIVATE "${IQM_ROOT}/include")
if(MSVC)
    target_compile_options(IQM-csf-reference-test PRIVATE /fp:strict /EHsc)
else()
    target_compile_options(IQM-csf-reference-test PRIVATE -ffp-contract=off)
endif()
add_executable(IQM-fft-codelet-test
    ../fft_codelet_gpu_test.cpp
    "${IQM_ROOT}/bin/shared/vulkan_res.cpp"
    "${IQM_ROOT}/bin/IQM/vulkan_instance.cpp"
    "${IQM_ROOT}/src/base/vulkan_runtime.cpp")
target_compile_definitions(IQM-fft-codelet-test PRIVATE VULKAN_HPP_NO_CONSTRUCTORS)
target_include_directories(IQM-fft-codelet-test PRIVATE
    "${IQM_ROOT}/include" "${IQM_ROOT}/bin/shared" "${IQM_ROOT}/bin/IQM")
target_include_directories(IQM-fft-codelet-test SYSTEM PRIVATE "${IQM_ROOT}/lib/stb")
target_link_libraries(IQM-fft-codelet-test PRIVATE Vulkan::Vulkan)
if(MSVC)
    target_compile_options(IQM-fft-codelet-test PRIVATE /EHsc)
endif()
if(WIN32)
    add_executable(IQM-fftw-plan-test ../inspect_fftw_plan.cpp)
    if(MSVC)
        target_compile_options(IQM-fftw-plan-test PRIVATE /EHsc)
    endif()
