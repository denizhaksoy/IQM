cmake_minimum_required(VERSION 3.29)
project(HDRVDP3GPUFFTSingle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(IQM_ROOT "../.." ABSOLUTE)
find_package(Vulkan REQUIRED)
add_executable(fft128-probe ../verify_fft128_single_gpu.cpp ../fft128_single_gpu.cpp
 ${IQM_ROOT}/src/base/vulkan_runtime.cpp ${IQM_ROOT}/bin/shared/vulkan_res.cpp
 ${IQM_ROOT}/bin/IQM/vulkan_instance.cpp)
target_include_directories(fft128-probe PRIVATE ${IQM_ROOT}/include ${IQM_ROOT}/bin/IQM ${IQM_ROOT}/bin/shared)
target_compile_definitions(fft128-probe PRIVATE VULKAN_HPP_NO_CONSTRUCTORS)
target_link_libraries(fft128-probe PRIVATE Vulkan::Vulkan)
if(MSVC)
 target_compile_options(fft128-probe PRIVATE /fp:strict /EHsc /W4)
endif()

get_target_property(convolution_sources fft128-probe SOURCES)
list(REMOVE_ITEM convolution_sources ../verify_fft128_single_gpu.cpp)
add_executable(fft128-convolution-probe ../verify_fft128_convolution_gpu.cpp ../fft_filter_single_gpu.cpp ../fft128_crop_single_gpu.cpp ${convolution_sources})
target_include_directories(fft128-convolution-probe PRIVATE ${IQM_ROOT}/include ${IQM_ROOT}/bin/IQM ${IQM_ROOT}/bin/shared)
target_compile_definitions(fft128-convolution-probe PRIVATE VULKAN_HPP_NO_CONSTRUCTORS)
target_link_libraries(fft128-convolution-probe PRIVATE Vulkan::Vulkan)
if(MSVC)
 target_compile_options(fft128-convolution-probe PRIVATE /fp:strict /EHsc /W4)
endif()
