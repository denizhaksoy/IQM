cmake_minimum_required(VERSION 3.29)
project(IQMRealHDRInputs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(prepare-real-hdr ../prepare_real_hdr.cpp)
add_executable(prepare-real-rgb ../prepare_real_rgb.cpp)
target_include_directories(prepare-real-rgb SYSTEM PRIVATE ../../lib/stb)
if(MSVC)
 target_compile_options(prepare-real-rgb PRIVATE /fp:strict /EHsc)
else()
 target_compile_options(prepare-real-rgb PRIVATE -ffp-contract=off)
endif()
target_include_directories(prepare-real-hdr SYSTEM PRIVATE ../../lib/stb)
if(MSVC)
 target_compile_options(prepare-real-hdr PRIVATE /fp:strict /EHsc)
else()
 target_compile_options(prepare-real-hdr PRIVATE -ffp-contract=off)
endif()
